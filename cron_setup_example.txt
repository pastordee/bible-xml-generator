# Cron Job Configuration for Bible Content Freshness Monitoring
# Add these lines to your crontab (run: crontab -e)

# Check content freshness daily at 6:00 AM
0 6 * * * cd /Users/prayercircle/Documents/bible/bible-xml-generator && .venv/bin/python monitor_content_freshness.py >> logs/freshness_monitor.log 2>&1

# Alternative: Check twice daily (6 AM and 6 PM)
# 0 6,18 * * * cd /Users/prayercircle/Documents/bible/bible-xml-generator && .venv/bin/python monitor_content_freshness.py >> logs/freshness_monitor.log 2>&1

# Weekly comprehensive check on Sundays at 2:00 AM
0 2 * * 0 cd /Users/prayercircle/Documents/bible/bible-xml-generator && .venv/bin/python check_content_freshness.py --output logs/weekly_report_$(date +\%Y\%m\%d).json >> logs/freshness_monitor.log 2>&1

# Setup Instructions:
# 1. Create logs directory: mkdir -p logs
# 2. Make scripts executable: chmod +x *.py
# 3. Add to crontab: crontab -e
